## Rulings this plan makes where Decision G is silent

- **Stage 4 runs `--kinds form,manage`, not `--kinds manage`.** The manage screen imports the form for its dialogs, and G2 refuses a kind whose dependencies are missing on disk, so G1's literal stage 4 command would be refused every time. Stage 5 still runs `--kinds form` as G1 says; after stage 4 it is a no-op (every file unchanged), and the prompt says so. The placeholder's marker line names the same `form,manage` command.
- **`--kinds` is one comma-separated string option** (`--kinds form,manage`), not a repeatable array, so the CLI, the skill and the mod's parser agree on one spelling. Spaces after commas are tolerated.
- **Identical means identical after `\r\n` to `\n`**, so a Windows checkout of a generated file does not refuse a rerun.
- **The hook denies creation, not rewrites.** G4 says "a Write that creates a generator-owned file"; a Write over an existing generated file passes the hook (the preserved-shape measure records it). Any `*.service.ts` under `src/app` counts as generator-owned, so a hand-written `theme.service.ts` would be denied; G4's list is followed as written.
- **The header constants are duplicated in the hook**, not imported from `schematics/`, so the hook stays self-contained like its siblings; a test fails when the two copies drift.
- **`createRoute` binds through `vm.createRoute`**, never as a literal in the template, so the rule that no spec value appears in markup still holds.
- **Created means `CREATE` or `UPDATE`** in the mod (except `app.routes.ts`), because a placeholder replacement and a `--force` overwrite print `UPDATE`. This widens `coverage.generatedFiles` too, which is why the series moves to 5.
- **`finish` re-reads schematic invocations from the stage streams**, as it already did for cost, so a fix to the reader reaches a refinished run and the end-to-end test needs no hand-built records.
- **The preserved shape regenerates from `spec.approved.json`** with the substrate's own `generateFiles` and `formatSource`, line-matched by longest common subsequence; Task 9 gives the reasons.
- **`generatedFromApprovedSpec` keeps its series 4 meaning** beside the new `reached`, so series 4 and series 5 meta files stay comparable.

'use strict';

/**
 * The `slice` schematic: reads one spec file and emits the kinds it lists, or the subset --kinds names.
 *
 *   ng generate slice --spec specs/hero.json [--kinds entity|form|manage[,...]] [--force]
 *
 * Plain CommonJS, no compile step. Deterministic: the same spec yields byte-identical files.
 *
 * Write policy, per file (harness/schematic-spec-s3-addendum.md Decision G2 on the launching branch):
 * - absent: created.
 * - identical to what the generator emits (line endings aside): skipped and logged as Unchanged.
 * - carrying the placeholder marker: overwritten without --force.
 * - anything else: added with tree.create(), so the CLI's workflow refuses the whole run and writes
 *   nothing; `--force` (a built-in `ng generate` flag the schematic never sees) lets it overwrite.
 * app.routes.ts is the one file that is edited rather than created.
 */

const { parseSpec } = require('./spec-schema.cjs');
const { buildContext, kebab, GENERATED_HEADER, PLACEHOLDER_MARKER } = require('./lib/naming.cjs');

const { form, formTemplate, formStyles } = require('./lib/emit-form.cjs');

const KIND_ORDER = ['entity', 'form', 'manage'];

/** What each kind imports from the others. */
const NEEDS = { entity: [], form: ['entity'], manage: ['entity', 'form'] };

/**
 * Pure: spec in, files out. Keys are workspace-relative paths in a fixed order.
 * @param {any} spec a spec that already passed parseSpec
 * @param {string[]} [kinds] the kinds to emit; defaults to every kind the spec lists
 * @returns {Map<string, string>}
 */
function generateFiles(spec, kinds = spec.kinds) {
  const ctx = buildContext(spec);
  const files = new Map();
  if (kinds.includes('entity')) {
    files.set(ctx.paths.model, entity.model(ctx));
    files.set(ctx.paths.schema, entity.schema(ctx));
    files.set(ctx.paths.service, entity.service(ctx));
    files.set(ctx.paths.example, entity.exampleData(ctx));
    files.set(ctx.paths.serviceSpec, entity.serviceSpec(ctx));
  }
  if (kinds.includes('form')) {
    files.set(ctx.paths.form, form(ctx));
    files.set(ctx.paths.formTemplate, formTemplate(ctx));
    files.set(ctx.paths.formStyles, formStyles(ctx));
  }
  if (kinds.includes('manage')) {
    files.set(ctx.paths.screen, manage.screen(ctx));
    files.set(ctx.paths.template, manage.template(ctx));
    files.set(ctx.paths.styles, manage.styles(ctx));
    files.set(ctx.paths.viewModel, manage.viewModel(ctx));
    files.set(ctx.paths.viewModelSpec, manage.viewModelSpec(ctx));
  }
  for (const [path, content] of files) {
    if (path.endsWith('.ts')) files.set(path, `${GENERATED_HEADER}\n${content}`);
  }
  return files;
}

/**
 * The kinds this run generates: the --kinds subset, in the fixed order, or every kind the spec lists.
 * @param {unknown} option the raw --kinds value
 * @param {any} spec
 * @param {string} specPath
 * @returns {string[]}
 */
function selectKinds(option, spec, specPath) {
  if (option === undefined || option === null || String(option).trim() === '') return [...spec.kinds];
  const asked = [...new Set(String(option).split(',').map((k) => k.trim()).filter(Boolean))];
  const unknown = asked.filter((k) => !KIND_ORDER.includes(k));
  if (unknown.length) {
    throw new SchematicsException(
      `--kinds names ${unknown.map((k) => `"${k}"`).join(', ')}, which is not a kind; the kinds are entity, form and manage.`,
    );
  }
  const unlisted = asked.filter((k) => !spec.kinds.includes(k));
  if (unlisted.length) {
    throw new SchematicsException(
      `--kinds names ${unlisted.join(', ')}, which ${specPath} does not list in its kinds (${spec.kinds.join(', ')}).`,
    );
  }
  return KIND_ORDER.filter((k) => asked.includes(k));
}

/**
 * A kind whose dependency is not generated in this run needs that dependency's files on disk.
 * Throws one readable message per missing dependency.
 */
function checkDependencies(tree, ctx, kinds, specPath) {
  const onDisk = { entity: [ctx.paths.model, ctx.paths.schema, ctx.paths.service], form: [ctx.paths.form] };
  const problems = [];
  for (const kind of kinds) {
    for (const dep of NEEDS[kind]) {
      if (kinds.includes(dep)) continue;
      const missing = onDisk[dep].filter((p) => !tree.exists(p));
      if (!missing.length) continue;
      const together = KIND_ORDER.filter((k) => k === dep || kinds.includes(k)).join(',');
      problems.push(
        `--kinds ${kind} needs the ${dep} kind's files on disk, and ${missing.join(', ')} ${missing.length === 1 ? 'is' : 'are'} missing. ` +
          `Generate them first (ng g slice --spec ${specPath} --kinds ${dep}) or in the same run (--kinds ${together}).`,
      );
    }
  }
  if (problems.length) throw new SchematicsException(problems.join('\n'));
}

function slice(options) {
  return async (tree, context) => {
    const specPath = String(options.spec).replace(/\\/g, '/').replace(/^\.\//, '');

    }
    let data;
    try {
      data = JSON.parse(buffer.toString('utf8'));
    } catch (e) {
      throw new SchematicsException(`Spec file ${specPath} is not valid JSON: ${e.message}`);
    }
    const parsed = parseSpec(data);
    if (!parsed.ok) {
      throw new SchematicsException(
        `Spec file ${specPath} is not valid:\n${parsed.errors.map((e) => `  - ${e}`).join('\n')}\n` +
          'It must satisfy schematics/slice/spec.schema.json.',
      );

    const spec = parsed.spec;
    const kinds = selectKinds(options.kinds, spec, specPath);
    const refProblems = checkRefTargets(tree, spec);
    if (refProblems.length) {
      const list = refProblems.map((e) => `  - ${e}`).join(String.fromCharCode(10));
      throw new SchematicsException(`Spec file ${specPath} has bad references:${String.fromCharCode(10)}${list}`);
    }
    const ctx = buildContext(spec);
    checkDependencies(tree, ctx, kinds, specPath);
    const files = generateFiles(spec, kinds);

    let routesSource = null;
    let routed = null;
    if (kinds.includes('manage')) {
      const existing = tree.read(ROUTES_FILE);
      if (!existing) throw new SchematicsException(`Cannot add the route: ${ROUTES_FILE} does not exist.`);
      try {
        routed = addRoute(existing.toString('utf8'), {
          path: spec.plural,
          importPath: `./${spec.plural}/${spec.plural}`,
          className: ctx.Plural,
        });
      } catch (e) {
        if (e instanceof RoutesError) throw new SchematicsException(e.message);
        throw e;
      }
      routesSource = routed.source;
    }

    const formatted = new Map();
    for (const [path, content] of files) formatted.set(path, await formatSource(path, content));
    if (routed && routed.added) routesSource = await formatSource(ROUTES_FILE, routesSource);

    for (const [path, content] of formatted) {
      const existing = tree.read(path);
      if (!existing) {
        tree.create(path, content);
        continue;
      }
      const text = existing.toString('utf8');
      if (text.replace(/\r\n/g, '\n') === content) context.logger.info(`Unchanged: ${path}`);
      else if (text.includes(PLACEHOLDER_MARKER)) tree.overwrite(path, content);
      else tree.create(path, content); // refused by the CLI unless --force, which overwrites
    }
    if (routed && routed.added) tree.overwrite(ROUTES_FILE, routesSource);
    else if (routed) context.logger.info(`Route "${spec.plural}" already exists in ${ROUTES_FILE}; left as is.`);
  };

#!/usr/bin/env node
// route-to-generator (Decision G4 of harness/schematic-spec-s3-addendum.md on the launching
// branch). One file, two entry points, chosen by the tool in the payload:
//
// Write (registered PreToolUse): denies a Write that creates a generator-owned file under src/app
// (a *.model.ts, *.schema.ts, *.service.ts, *.service.spec.ts or *-example-data.ts, or anything
// under ui/<name>-form/), and a Write over any file under src/ that carries the placeholder
// marker. The message names the spec and the command. Edit is not matched: filling a generated
// file in place is allowed, and the other gates police what goes into it.
//
// Bash or PowerShell (registered PostToolUse): a shell can create any file without a Write. This
// asks git which generator-owned TypeScript files under src/app are new or modified and reports
// each one whose first line does not carry the generator's header, because the generator did not
// write it.
//
// Registered only by harness/gate/s4-schematic.settings.json on the launching branch, so it runs
// in the schematic condition only. Fails closed on an unparseable payload, an unreadable file or a
// git failure.

import { spawnSync } from 'node:child_process';
import { existsSync, readFileSync } from 'node:fs';
import { dirname, join, relative, resolve } from 'node:path';
import { fileURLToPath, pathToFileURL } from 'node:url';
import { logFiring } from './_hook-log.mjs';

const ROOT = join(dirname(fileURLToPath(import.meta.url)), '..', '..');
const HOOK = 'route-to-generator';

// Kept equal to GENERATED_MARKER and PLACEHOLDER_MARKER in schematics/slice/lib/naming.cjs.
// harness/gate/route-to-generator-hook.test.mjs fails when they drift.
export const GENERATED_MARKER = '@generated by the slice schematic';
export const PLACEHOLDER_MARKER = 'slice:placeholder';

/** Which kind owns a repo-relative path, and the entity it belongs to; null when the generator does not own it. */
export function ownerOf(rel) {
  if (!rel.startsWith('src/app/')) return null;
  const form = /^src\/app\/ui\/([a-z0-9-]+)-form\//.exec(rel);
  if (form) return { kind: 'form', entity: form[1] };
  const entity = /\/([^/]+?)(\.model\.ts|\.schema\.ts|\.service\.spec\.ts|\.service\.ts|-example-data\.ts)$/.exec(rel);
  if (entity) return { kind: 'entity', entity: entity[1] };
  return null;
}

const command = (owner) => `npx ng g slice --spec specs/${owner.entity}.json --kinds ${owner.kind}`;

/** The denial message for a Write, or null when the Write may go ahead. */
export function writeDenial(rel, exists, text) {
  if (!rel.startsWith('src/')) return null;
  if (exists && text.includes(PLACEHOLDER_MARKER)) {
    const line = text.split(/\r?\n/).find((l) => l.includes(PLACEHOLDER_MARKER)).trim();
    return (
      `${rel} is a placeholder the slice generator owns (${PLACEHOLDER_MARKER}). Do not write over it: ` +
      `generate it from the approved spec. Its marker line names the command:\n  ${line}`
    );
  }
  const owner = exists ? null : ownerOf(rel);
  if (!owner) return null;
  return (
    `${rel} is generator-owned (the ${owner.kind} kind). Do not write it by hand: generate it from the approved spec with\n` +
    `  ${command(owner)}\n` +
    'then change the generated file in place with Edit. A different shape goes through the spec and --force.'
  );
}

function changedOwnedTypeScript() {
  const r = spawnSync('git', ['-C', ROOT, 'status', '--porcelain=v1', '-z', '-uall', '--', 'src/app'], { encoding: 'utf8' });
  if (r.status !== 0) throw new Error(`git status failed: ${String(r.stderr).trim()}`);
  const parts = r.stdout.split('\0').filter(Boolean);
  const out = [];
  for (let i = 0; i < parts.length; i++) {
    const status = parts[i].slice(0, 2);
    const path = parts[i].slice(3);
    if (status.startsWith('R') || status.startsWith('C')) i++; // -z puts the source path next
    if (status.includes('D')) continue;
    out.push(path);
  }
  return out.filter((p) => p.endsWith('.ts') && ownerOf(p) && existsSync(join(ROOT, p)));
}

function fail(message) {
  process.stderr.write(`${HOOK}: ${message}; failing closed.\n`);
  process.exit(2);
}

function onWrite(payload) {
  const file = payload.tool_input?.file_path ?? payload.tool_input?.filePath ?? null;
  if (!file) process.exit(0);
  const abs = resolve(String(file));
  const rel = relative(ROOT, abs).replaceAll('\\', '/');
  const exists = existsSync(abs);
  let text = '';
  if (exists) {
    try {
      text = readFileSync(abs, 'utf8');
    } catch (err) {
      fail(`could not read ${rel} (${err?.message ?? err})`);
    }
  }
  const message = writeDenial(rel, exists, text);
  if (!message) process.exit(0);
  logFiring(HOOK, rel, [message], [HOOK]);
  process.stderr.write(`Route-to-generator gate blocked this Write.\n${message}\n`);
  process.exit(2);
}

function onShell() {
  let files;
  try {
    files = changedOwnedTypeScript();
  } catch (err) {
    fail(String(err?.message ?? err));
  }
  const headerless = files.filter((p) => !readFileSync(join(ROOT, p), 'utf8').split(/\r?\n/, 1)[0].includes(GENERATED_MARKER));
  if (headerless.length === 0) process.exit(0);
  const messages = headerless.map(
    (p) => `${p} is generator-owned but does not start with the generator's header, so it was written by hand. Delete it and generate it: ${command(ownerOf(p))}`,
  );
  logFiring(HOOK, 'src/app/', messages, [HOOK]);
  process.stderr.write(
    `Route-to-generator gate: ${headerless.length} generator-owned file(s) without the generator's header.\n` +
      messages.map((m) => `  ${m}`).join('\n') +
      `\nEvery generated TypeScript file starts with "// ${GENERATED_MARKER} ...". Fix the above before continuing.\n`,
  );
  process.exit(2);
}

async function main() {
  let raw = '';
  for await (const chunk of process.stdin) raw += chunk;
  let payload;
  try {
    payload = JSON.parse(raw);
    if (payload === null || typeof payload !== 'object') throw new Error('payload is not an object');
  } catch {
    fail('could not parse hook payload');
  }
  const tool = payload.tool_name ?? '';
  if (tool === 'Write') onWrite(payload);
  else if (tool === 'Bash' || tool === 'PowerShell') onShell();
  else process.exit(0);
}

if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  main().catch((err) => fail(String(err?.message ?? err)));
}

/** Run one arm.mjs subcommand; `out` is the JSON object on its last stdout line. */
export function arm(command, args) {
  const r = spawnSync(process.execPath, [ARM, command, ...args], { encoding: 'utf8' });
  const lines = String(r.stdout).trim().split('\n').filter(Boolean);
  let out = null;
  try {
    out = lines.length ? JSON.parse(lines.at(-1)) : null;
  } catch {
    out = { unparsed: lines.at(-1) };
  }
  return { status: r.status, out, stderr: r.stderr };
}

/** A stage stream as `claude --output-format stream-json` writes it: init, one Bash call and its output, the result. */

  const lines = [{ type: 'system', subtype: 'init', model: 'claude-haiku-5-5', plugins: [], mcp_servers: [], output_style: 'default', tools: [] }];

    lines.push({ type: 'assistant', message: { content: [{ type: 'tool_use', id: `t${n}`, name: 'Bash', input: { command } }] } });
    lines.push({ type: 'user', message: { content: [{ type: 'tool_result', tool_use_id: `t${n}`, content: output }] } });
  }
  lines.push({ type: 'result', subtype: 'success', is_error: false, num_turns: 1, total_cost_usd: 0.01, duration_ms: 10, duration_api_ms: 5 });
  writeFileSync(join(work, `stage-${n}.jsonl`), lines.map((l) => JSON.stringify(l)).join('\n') + '\n');
}

Why regenerate, not record: the preserved share needs the generator's output line by line. A hash recorded at generation time only says whether a whole file changed, and a snapshot taken at the end of the generating stage already includes that stage's own edits. The generator is deterministic and pure (`generateFiles(spec)` is spec in, files out, and the golden test holds it byte-exact), the approved spec is saved in the evidence, and `schematics/` is protected, so at finish the substrate's own `generateFiles` plus `formatSource` on `spec.approved.json` reproduces exactly what `ng g slice` wrote. No temp copy of the tree is needed, because `generateFiles` reads nothing from disk. Lines are matched by longest common subsequence per file, after `\r\n` to `\n`, so an insertion costs nothing and every changed or deleted generator line costs one. A generated file that is missing at its generated path at the end (moved, renamed or deleted) preserves zero lines. Only the kinds reached are counted, since a kind never generated has no generated lines. A spec changed after approval and regenerated with `--force` counts as changed lines; the drift record says why.

// Absolute and case-folded when the run root is known, so ./specs/hero.json, specs\hero.json and

function approvedRuns(record, specPath, root) {
  const key = (f) => (root ? posix(resolve(root, f)).toLowerCase() : normPath(f));
  const want = key(specPath);
  return (record.schematicInvocations ?? []).filter((i) => {

    return v !== undefined && key(v) === want;

// Series 4 bar 3, kept for continuity: one stage generated the manage screen and the form from the approved spec.
export function generatedFromApprovedSpec(record, specPath, approvedSpec, root) {
  if (!specPath || !approvedSpec?.plural || !approvedSpec?.entity) return false;
  const runs = approvedRuns(record, specPath, root);
  if (!runs.length) return false;
  // Per invocation; an older record without per-invocation files falls back to the stage union.
  const created = runs.flatMap((i) => i.created ?? record.schematicCreated ?? []).map(normPath);
  const dirs = kindDirs(approvedSpec);
  return created.some((f) => tsFileIn(f, dirs.manage)) && created.some((f) => tsFileIn(f, dirs.form));
}

// Series 5 bar 3 (Decision G5): a kind is reached when a real run on the approved spec asked for
// it (--kinds, or the spec's kinds when the command names none) and its CREATE or UPDATE lines
// include a TypeScript file in that kind's folder. Output the subject cut off (| tail) shows no
// such lines, so that run reaches nothing: the stream is the evidence.
export function reachedKinds(records, specPath, approvedSpec, root) {
  const out = { entity: false, form: false, manage: false, all: false, stages: { entity: [], form: [], manage: [] } };
  if (!specPath || !approvedSpec?.plural || !approvedSpec?.entity) return out;
  const dirs = kindDirs(approvedSpec);
  const specKinds = Array.isArray(approvedSpec.kinds) ? approvedSpec.kinds : KINDS;
  for (const record of records) {
    for (const run of approvedRuns(record, specPath, root)) {
      const asked = parseKindsOption(run.command) ?? specKinds;
      const created = (run.created ?? record.schematicCreated ?? []).map(normPath);
      for (const kind of KINDS) {
        if (!asked.includes(kind) || !created.some((f) => tsFileIn(f, dirs[kind]))) continue;
        out[kind] = true;
        if (!out.stages[kind].includes(record.name)) out.stages[kind].push(record.name);
      }
    }
  }
  out.all = KINDS.every((k) => out[k]);
  return out;
}

// Length of the longest common subsequence of two line lists: the generator lines that survive, in order.
export function lcsLength(a, b) {
  let prev = new Array(b.length + 1).fill(0);
  for (let i = 1; i <= a.length; i++) {
    const cur = new Array(b.length + 1).fill(0);
    for (let j = 1; j <= b.length; j++) cur[j] = a[i - 1] === b[j - 1] ? prev[j - 1] + 1 : Math.max(prev[j], cur[j - 1]);
    prev = cur;
  }
  return prev[b.length];
}

// The share of generated lines still byte-identical to the generator's output (line endings aside).
// A file missing at its generated path preserves nothing.
export function preservedShape(expected, readActual) {
  const files = {};
  let generatedLines = 0;
  let preservedLines = 0;
  for (const [path, want] of expected) {
    const a = linesOf(want);
    const actual = readActual(path);
    const preserved = actual === null ? 0 : lcsLength(a, linesOf(actual));
    files[path] = { lines: a.length, preserved, present: actual !== null };
    generatedLines += a.length;
    preservedLines += preserved;
  }
  return { generatedLines, preservedLines, share: generatedLines ? Number((preservedLines / generatedLines).toFixed(4)) : null, files };
}

// What `ng g slice` writes for the approved spec, from the substrate's own generator at the run
// root: generateFiles is pure and the golden test holds it byte-exact, so this is the generator's
// output without re-running the CLI or copying the tree.
export async function expectedGeneratorOutput(root, approvedSpec) {

  const out = new Map();
  for (const [path, content] of generateFiles(parsed.spec)) out.set(path, await formatSource(path, content));
  return out;
}
